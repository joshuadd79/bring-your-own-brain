# =====================================================================
# bringyourownbrain.org  -  .htaccess   (Apache 2.4 / LiteSpeed)
# Upload to the web root and rename htaccess.txt -> .htaccess
#
# Needs: mod_rewrite, mod_headers, mod_mime, mod_setenvif (all standard)
# Purpose:
#   * connect.html can be embedded in an iframe by ANY site
#   * byob-web-sdk.js can be loaded by <script src> from ANY site
#   * everything else on the domain cannot be framed
#   * correct MIME types (nosniff is on, so wrong types = blocked scripts)
# =====================================================================

Options -Indexes
AddDefaultCharset UTF-8

# ---------- MIME types ----------
<IfModule mod_mime.c>
  AddType application/javascript .js .mjs
  AddType application/wasm       .wasm
  AddType application/json       .json
  AddType text/html              .html
</IfModule>

# ---------- HTTPS + one canonical host ----------
# (storage access is keyed to the exact site, so www and non-www must not both serve)
<IfModule mod_rewrite.c>
  RewriteEngine On

  RewriteCond %{HTTPS} !=on
  RewriteCond %{HTTP:X-Forwarded-Proto} !https [NC]
  RewriteRule ^ https://bringyourownbrain.org%{REQUEST_URI} [L,R=301]

  RewriteCond %{HTTP_HOST} ^www\. [NC]
  RewriteRule ^ https://bringyourownbrain.org%{REQUEST_URI} [L,R=301]
</IfModule>

# ---------- Which URLs are public ----------
# BYOB_EMBED  = pages other sites may put in an iframe
<IfModule mod_setenvif.c>
  SetEnvIf Request_URI "^/(connect\.html|connect/.*)$" BYOB_EMBED=1
</IfModule>

<IfModule mod_headers.c>

  # ---- Site-wide hardening ----
  Header always set X-Content-Type-Options "nosniff"
  Header always set Referrer-Policy "strict-origin-when-cross-origin"
  # Uncomment once you are sure HTTPS works everywhere:
  # Header always set Strict-Transport-Security "max-age=31536000"

  # ---- Everything EXCEPT the hub: nobody else may frame it ----
  Header always set Content-Security-Policy "frame-ancestors 'self'" env=!BYOB_EMBED
  Header always set X-Frame-Options "SAMEORIGIN"                      env=!BYOB_EMBED

  # ---- The hub: ANY site may embed it ----
  # (a <meta> CSP cannot do frame-ancestors, it must be an HTTP header)
  Header always set   Content-Security-Policy "frame-ancestors *" env=BYOB_EMBED
  Header always unset X-Frame-Options                              env=BYOB_EMBED

  # ---- The SDK: loadable by <script src> from any domain ----
  <FilesMatch "^byob-web-sdk\.js$">
    Header set Access-Control-Allow-Origin     "*"
    Header set Cross-Origin-Resource-Policy    "cross-origin"
    Header set Cache-Control                   "public, max-age=300"
  </FilesMatch>

  # ---- Hub + engine: always revalidate so fixes reach everyone quickly ----
  <FilesMatch "^(connect\.html|offscreen\.js)$">
    Header set Cache-Control "no-cache"
  </FilesMatch>

  # ---- Bundled engine files (vendor/): safe to cache for a week ----
  <FilesMatch "\.(wasm|mjs)$">
    Header set Cache-Control "public, max-age=604800"
  </FilesMatch>

  # DO NOT add these on the hub, they break embedding / storage access / WebGPU:
  #   Cross-Origin-Embedder-Policy, Cross-Origin-Opener-Policy,
  #   a restrictive Permissions-Policy, or X-Frame-Options.

</IfModule>

# ---------- Compression ----------
<IfModule mod_deflate.c>
  AddOutputFilterByType DEFLATE text/html text/css application/javascript application/json application/wasm
</IfModule>

# ---------- Keep junk private ----------
<FilesMatch "\.(bak|sql|log|ini|env|old)$">
  Require all denied
</FilesMatch>
